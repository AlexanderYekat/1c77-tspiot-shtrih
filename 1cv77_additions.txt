Функция JSONСтрока(Значение)
	К = Симв(34);
	Стр = СокрЛП(Значение);
	Стр = СтрЗаменитьСтр, "\", "\\");
	Стр = СтрЗаменить(Стр, Симв(34), "\" + Симв(34));
	Стр = СтрЗаменить(Стр, Симв(13), "\r");
	Стр = СтрЗаменить(Стр, Симв(10), "\n");
	Стр = СтрЗаменить(Стр, Симв(9), "\t");
	// ASCII GS для маркировки. В JSON нельзя передавать "сырой" управляющий символ.
	Стр = СтрЗаменить(Стр, Симв(29), "\u001D");
	Возврат К + Стр + К;
КонецФункции

Функция JSONЧисло(Значение)
	Если ПустоеЗначение(Значение) = 1 Тогда
		Возврат "0";
	КонецЕсли;

	Стр = СокрЛП(Формат(Значение, "Ч20.4"));
	Стр = СтрЗаменить(Стр, " ", "");
	Стр = СтрЗаменить(Стр, ",", ".");

	Пока (СтрДлина(Стр) > 0) И (Прав(Стр, 1) = "0") И (Найти(Стр, ".") > 0) Цикл
		Стр = Лев(Стр, СтрДлина(Стр) - 1);
	КонецЦикла;
	Если Прав(Стр, 1) = "." Тогда
		Стр = Лев(Стр, СтрДлина(Стр) - 1);
	КонецЕсли;
	Если Стр = "" Тогда
		Стр = "0";
	КонецЕсли;

	Возврат Стр;
КонецФункции

	К = Симв(34);

Функция ПодготовитьМаркуДляККТ(Марка)
	// Повторяет старую логику из ПечататьЧек():
	// Сред(Марка,1,31) + GS + Сред(Марка,36,6) + GS + Сред(Марка,46)
	Если ПустаяСтрока(Марка) = 1 Тогда
		Возврат "";
	КонецЕсли;

	Возврат СокрЛП(Сред(Марка, 1, 31) + Симв(29) + Сред(Марка, 36, 6) + Симв(29) + Сред(Марка, 46));

Функция HTTPЗапросJSON(Метод, Адрес, Тело, Ответ, HTTPСтатус)
	Рез = 0;
	Ответ = "";
	HTTPСтатус = 0;

	Попытка
		HTTP = СоздатьОбъект("WinHttp.WinHttpRequest.5.1");
		HTTP.Open(Метод, Адрес, 0);
		HTTP.SetTimeouts(5000, 5000, 120000, 120000);
		HTTP.SetRequestHeader("Content-Type", "application/json; charset=utf-8");
		HTTP.SetRequestHeader("Accept", "application/json");
		HTTP.Send(Тело);
		HTTPСтатус = HTTP.Status;
		Ответ = HTTP.ResponseText;
		Если (HTTPСтатус >= 200) И (HTTPСтатус < 300) Тогда
			Рез = 1;
		Иначе
			ОписаниеРезультата = "HTTP " + СокрЛП(HTTPСтатус) + ": " + Ответ;
		КонецЕсли;
	Исключение
		ОписаниеРезультата = "Ошибка HTTP-запроса к proxy-1c-fmuapi: " + ОписаниеОшибки();
		Рез = 0;
	КонецПопытки;

	Возврат Рез;
КонецФункции

Функция JSONПолучитьЗначение(Текст, Имя)
	К = Симв(34);
	Шаблон = К + Имя + К;
	П = Найти(Текст, Шаблон);
	Если П = 0 Тогда
		Возврат "";
	КонецЕсли;

	Остаток = Сред(Текст, П + СтрДлина(Шаблон));
	ПДв = Найти(Остаток, ":");
	Если ПДв = 0 Тогда
		Возврат "";
	КонецЕсли;

	Остаток = СокрЛ(Сред(Остаток, ПДв + 1));
	Если Лев(Остаток, 1) = К Тогда
		Остаток = Сред(Остаток, 2);
		Рез = "";
		Пока СтрДлина(Остаток) > 0 Цикл
			Сим = Лев(Остаток, 1);
			Если Сим = К Тогда
				Прервать;
			КонецЕсли;
			Рез = Рез + Сим;
			Остаток = Сред(Остаток, 2);
		КонецЦикла;
		Возврат Рез;
	КонецЕсли;

	Рез = "";
	Пока СтрДлина(Остаток) > 0 Цикл
		Сим = Лев(Остаток, 1);
		Если (Сим = ",") Или (Сим = "}") Или (Сим = "]") Тогда
			Прервать;
		КонецЕсли;
		Рез = Рез + Сим;
		Остаток = Сред(Остаток, 2);
	КонецЦикла;

	Возврат СокрЛП(Рез);
КонецФункции

Функция ПолучитьИННиФНКассыЧерезВнешнююПрограмму()
	Ответ = "";
	HTTPСтатус = 0;
	Если HTTPЗапросJSON("GET", ПроксиАдресККТ("/kkt/info"), "", Ответ, HTTPСтатус) = 0 Тогда
		Возврат "";
	КонецЕсли;

	Если Число(JSONПолучитьЗначение(Ответ, "result")) <> 1 Тогда
		ОписаниеРезультата = JSONПолучитьЗначение(Ответ, "description");
		Если ОписаниеРезультата = "" Тогда
			ОписаниеРезультата = Ответ;
		КонецЕсли;
		Возврат "";
	КонецЕсли;

	// ВАЖНО: если ваша старая ПолучитьИННиФНКассы() возвращала другую структуру,
	// оставьте имена полей прежними. Здесь универсальный вариант через СписокЗначений.
	Сп = СоздатьОбъект("СписокЗначений");
	Сп.Установить("ИНН", JSONПолучитьЗначение(Ответ, "inn"));
	Сп.Установить("ФН", JSONПолучитьЗначение(Ответ, "fnNumber"));
	Сп.Установить("inn", JSONПолучитьЗначение(Ответ, "inn"));
	Сп.Установить("fn", JSONПолучитьЗначение(Ответ, "fnNumber"));
	Сп.Установить("serialNumber", JSONПолучитьЗначение(Ответ, "serialNumber"));

	Возврат Сп;
КонецФункции

Функция СобратьJSONЧекаДляВнешнейПечати(Пароль, СписокТоваров, Получено, ПризнВозврата, НомерСекции, НомерФР, Оплата, ИтогДК, НП)
	К = Симв(34);
	ТоварыJSON = "";
	Первый = 1;

	СписокТоваров.ВыбратьСтроки();
	Пока СписокТоваров.ПолучитьСтроку() = 1 Цикл
		Если Первый = 0 Тогда
			ТоварыJSON = ТоварыJSON + ",";
		КонецЕсли;
		Первый = 0;

		МаркаККТ = "";
		Если ПустаяСтрока(СписокТоваров.Маркировка) = 0 Тогда
			МаркаККТ = ПодготовитьМаркуДляККТ(СписокТоваров.Маркировка);
		КонецЕсли;

		СтрокаТовара = "{";
		СтрокаТовара = СтрокаТовара + JSONПараСтрока("name", СписокТоваров.ТовНаим) + ",";
		СтрокаТовара = СтрокаТовара + JSONПараЧисло("quantity", СписокТоваров.Количество) + ",";
		// price — именно то значение, которое раньше передавалось в Объект.Price.
		СтрокаТовара = СтрокаТовара + JSONПараЧисло("price", СписокТоваров.Сумма) + ",";
		СтрокаТовара = СтрокаТовара + JSONПараЧисло("basePrice", СписокТоваров.Цена) + ",";
		СтрокаТовара = СтрокаТовара + JSONПараЧисло("discount", СписокТоваров.Дисконт) + ",";
		СтрокаТовара = СтрокаТовара + JSONПараЧисло("pv", СписокТоваров.ПВ) + ",";
		СтрокаТовара = СтрокаТовара + JSONПараЧисло("department", НомерСекции) + ",";
		СтрокаТовара = СтрокаТовара + JSONПараСтрока("markRaw", СписокТоваров.Маркировка) + ",";
		СтрокаТовара = СтрокаТовара + JSONПараСтрока("markKkt", МаркаККТ) + ",";
		СтрокаТовара = СтрокаТовара + JSONПараСтрока("uuid", СписокТоваров.UUID) + ",";
		СтрокаТовара = СтрокаТовара + JSONПараСтрока("time", СписокТоваров.time);
		СтрокаТовара = СтрокаТовара + "}";

		ТоварыJSON = ТоварыJSON + СтрокаТовара;
	КонецЦикла;

	Тело = "{";
	Тело = Тело + JSONПараСтрока("command", "print_check") + ",";
	Тело = Тело + JSONПараЧисло("password", Пароль) + ",";
	Тело = Тело + JSONПараЧисло("returnCheck", ПризнВозврата) + ",";
	Тело = Тело + JSONПараЧисло("department", НомерСекции) + ",";
	Тело = Тело + JSONПараЧисло("payment", Оплата) + ",";
	Тело = Тело + JSONПараЧисло("received", Получено) + ",";
	Тело = Тело + JSONПараЧисло("total", СписокТоваров.Итог("Сумма")) + ",";
	Тело = Тело + JSONПараЧисло("discountTotal", ИтогДК) + ",";
	Тело = Тело + JSONПараСтрока("sellerCode", НП) + ",";
	Тело = Тело + JSONПараСтрока("fiscalRegisterHint", НомерФР) + ",";
	Тело = Тело + К + "items" + К + ":[" + ТоварыJSON + "]";
	Тело = Тело + "}";

	Возврат Тело;
КонецФункции

Функция ПечататьЧекЧерезВнешнююПрограмму(Объект, Пароль, СписокТоваров, Получено, ПризнВозврата, НомерСекции, НомерЧека, НомерФР, Оплата, ИтогДК, НП)
	Ответ = "";
	HTTPСтатус = 0;
	Тело = СобратьJSONЧекаДляВнешнейПечати(Пароль, СписокТоваров, Получено, ПризнВозврата, НомерСекции, НомерФР, Оплата, ИтогДК, НП);

	Сообщить("Передача чека во внешнюю программу печати через proxy-1c-fmuapi...");
	ЗаписьЖурналаРегистрации("print-check request: " + Тело);

	Если HTTPЗапросJSON("POST", ПроксиАдресККТ("/kkt/print-check"), Тело, Ответ, HTTPСтатус) = 0 Тогда
		Возврат 0;
	КонецЕсли;

	ЗаписьЖурналаРегистрации("print-check response: " + Ответ);

	Если Число(JSONПолучитьЗначение(Ответ, "result")) <> 1 Тогда
		ОписаниеРезультата = JSONПолучитьЗначение(Ответ, "description");
		Если ОписаниеРезультата = "" Тогда
			ОписаниеРезультата = Ответ;
		КонецЕсли;
		Возврат 0;
	КонецЕсли;

	НомерЧека = Число(JSONПолучитьЗначение(Ответ, "checkNumber"));
	НомерФР = JSONПолучитьЗначение(Ответ, "serialNumber");
	ОписаниеРезультата = JSONПолучитьЗначение(Ответ, "description");
	Возврат 1;
КонецФункции
