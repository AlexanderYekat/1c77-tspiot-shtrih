Функция РазобратьСтрокуCSV(Стр)
    Результат = СоздатьОбъект("СписокЗначений");
    ТекущееЗначение = "";
    ВКавычках = 0;   
    Для Н = 1 По СтрДлина(Стр) Цикл
        Символ = Сред(Стр, Н, 1);       
        Если Символ = """" Тогда      
			//обрбатываем двойные кавчки
            ВКавычках = 1 - ВКавычках;
			Если ВКавычках = 1 тогда
				Если ТекущееЗначение <> "" тогда
					ТекущееЗначение = ТекущееЗначение + Символ;
				конецЕсли;				
			конецЕсли;
            Продолжить;
        КонецЕсли;        
        Если (Символ = ",") И (НЕ (ВКавычках = 1)) Тогда
            Результат.ДобавитьЗначение(ТекущееЗначение);
            ТекущееЗначение = "";
            Продолжить;
        КонецЕсли;        
        ТекущееЗначение = ТекущееЗначение + Символ;
    КонецЦикла;
    // Добавляем последнее значение
    Результат.ДобавитьЗначение(ТекущееЗначение);    
    Возврат Результат;
КонецФункции //РазобратьСтрокуCSV
