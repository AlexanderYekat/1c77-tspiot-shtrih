функция ПроверитьМаркуРазрешительныйРежим(параметрыСервиса, данныеЗапроса, кодОшибки, ОписаниеРезультата = "")
	резСпс = СоздатьОбъект("СписокЗначений");                                                                                   
	кодОшибки = 0;                                                                                                            	
	марка = данныеЗапроса.Получить("макра");

	ФН = данныеЗапроса.Получить("ФН");
	ОписаниеРезультатаПодлкючениеКомпоненты = "";
	резПодключенияКомпоненты = ПодключитьV7PlusКомпоненту(глСоединение, ОписаниеРезультатаПодлкючениеКомпоненты, 1);
	Если резПодключенияКомпоненты = 0 тогда
		кодОшибки = 1001;
		ОписаниеРезультата = ОписаниеРезультата + ОписаниеРезультатаПодлкючениеКомпоненты;
		возврат резСпс;
	конецЕсли;;     		
	json = сформироватьJSON(марка, ИНН, ФН);
	кодОшибкиПроверкиМарки = 0; ОписаниеРезультатаПроверкиМакри = ""; ВыводитьОписаниеОшибок = 0;
	JSONРезультатПроверки = ОтправитьМаркуНаПроверку(параметрыСервиса.Получить("ip"), параметрыСервиса.Получить("port"), параметрыСервиса.Получить("apiPath"), json, кодОшибкиПроверкиМарки, ОписаниеРезультатаПроверкиМакри, ВыводитьОписаниеОшибок);
	Если кодОшибки <> 0 тогда
		кодОшибки = 1000+кодОшибкиПроверкиМарки;
		ОписаниеРезультата = ОписаниеРезультата + ОписаниеРезультатаПроверкиМакри;
		возврат резСпс;
	конецЕсли;
	спПолейОтвета = unMarshalLimit(JSONРезультатПроверки);
	codeFMU = спПолейОтвета.Получить("Code");
	codeChZnak = спПолейОтвета.Получить("truemark_response");
	UUID = спПолейОтвета.Получить("reqId");
	time = спПолейОтвета.Получить("reqTimestamp");	
	
	таблПолейПроверки = СформироватьСтрукутруТаблицыШаблонаПроверкиМарки();
	ПолучитьШаблонПроверкиМарки(таблПолейПроверки);
	ЗаполнитьПоляТаблицыПоОтветуJSON(таблПолейПроверки, JSONРезультатПроверки);
	
	резСпс.Установить("Code", codeFMU);
	резСпс.Установить("Code честный знак", codeChZnak);
	резСпс.Установить("UUID",  UUID);
	резСпс.Установить("time", time);
	таблПолейПроверки.ВыбратьСтроки();
	Пока таблПолейПроверки.ПолучитьСтроку() = 1 цикл
		резСпс.Установить(таблПолейПроверки.поле, таблПолейПроверки.value);

	возврат резСпс;

    // Символ ASCII 29 (Group Separator)
	//0104810268041853212pXWZkFn\u001d93pmiS   
	//0102900252571014215es.GCSfx>qj291EE0992+lqtVPbyjVsbfCzISFJ8BeGaJbgbBMs550mTgR6fOqQ=	
	кодОшибки = 0;

	КодРазделителяU = "\u001d";

    // Проверяем минимальную длину входной строки (16 + 15 = 31 символ для сокращенного формата)
    Если СтрДлина(Маркировка) < 31 Тогда

        Сообщить("Ошибка: Некорректная длина строки маркировки!");

    ПолныйФормат = ЕстьПолныйФормат(Маркировка);
    
    // Текущая позиция в строке
    ТекущаяПозиция = 1;
    
    // Извлекаем первую часть (01 + 14 цифр = 16 символов)
    ЧастьСтроки1 = Лев(Маркировка, 16);
	ТекущаяПозиция = 16;
	                            
	длинаСерии = 13; //длинаСерии = 8
	позНайд93 = Найти(Сред(Маркировка, ТекущаяПозиция + 1, 15), "93");
	Если позНайд93 > 0 тогда
		длинаСерии = 8;
	конецЕсли;                       
	всегоВырез = длинаСерии + 2;
    // Извлекаем вторую часть (21 + 13 символов = 15 символов)
    ЧастьСтроки2 = Сред(Маркировка, ТекущаяПозиция + 1, всегоВырез);

        //ЧастьСтроки2 = Лев(ЧастьСтроки2, СтрДлина(ЧастьСтроки2) - 1);
        ТекущаяПозиция = ТекущаяПозиция + всегоВырез + 1;

	КодСимволаРазделителя = 0;

Функция ДекодироватьBase64(Текст64)
    Перем Алфавит;
    Перем Группа, Символ;
    Перем Значение, Разряд;
    Перем Символ6, Значение6;
    Перем Значение8, Символ8;
    Перем Текст;
    
    // Подготовка алфавита и результата
    Алфавит = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/=";
    Текст = "";
    
    // Получение групп по 4 значащих символа
    Группа = "";
    Для Номер = 1 По СтрДлина(Текст64) Цикл
        Символ = Сред(Текст64, Номер, 1);
        Если Найти(Алфавит, Символ) = 0 Тогда
            Продолжить;
        КонецЕсли;
        Группа = Группа + Символ;
        Если СтрДлина(Группа) = 4 Тогда


            // Получение числового значения из 4-х символов
            Значение = 0;
            Разряд = 3;
            Пока Разряд >= 0 Цикл
                Символ6 = Сред(Группа, 4 - Разряд, 1);
                Если Символ6 = "=" Тогда
                    Значение6 = 0;
                Иначе
                    Значение6 = Найти(Алфавит, Символ6) - 1;
                КонецЕсли;
                Если Разряд = 1 Тогда
                    Значение6 = Значение6 * 64;
                ИначеЕсли Разряд = 2 Тогда
                    Значение6 = Значение6 * 64 * 64;
                ИначеЕсли Разряд = 3 Тогда
                    Значение6 = Значение6 * 64 * 64 * 64;
                КонецЕсли;
                Значение = Значение + Значение6;
                Разряд = Разряд - 1;
            КонецЦикла;


            // Получение 3-х символов из числового значения            
            Разряд = 2;
            Пока Разряд >= 0 Цикл
                Значение8 = Значение;
                Если Разряд = 2 Тогда
                    Значение8 = Цел(Значение8 / 256 / 256);
                    Значение = Значение - Значение8 * 256 * 256;
                ИначеЕсли Разряд = 1 Тогда
                    Значение8 = Цел(Значение8 / 256);
                    Значение = Значение - Значение8 * 256;
                КонецЕсли;
                
                Если Значение8 <>  0 Тогда // [+] Decker
                    // чтобы не добавлялись лишние символы пробела
                    Символ8 = Симв(Значение8);
                    Текст = Текст + Символ8;
                КонецЕсли;
                
                Разряд = Разряд - 1;
            КонецЦикла;
            
            Группа = "";
        КонецЕсли;
    КонецЦикла;
    
    Возврат Текст;
    
КонецФункции // ДекодироватьBase64()
